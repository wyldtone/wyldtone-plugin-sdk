cmake_minimum_required(VERSION 3.16)
project(wyldtone_yin_tracker LANGUAGES CXX)
add_library(wyldtone_yin STATIC YinTracker.cpp)
target_compile_features(wyldtone_yin PUBLIC cxx_std_17)
target_include_directories(wyldtone_yin PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(wyldtone_yin PRIVATE /W4)
else()
  target_compile_options(wyldtone_yin PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(yin_offline YinOffline.cpp)
target_link_libraries(yin_offline PRIVATE wyldtone_yin)
enable_testing()
add_executable(yin_tracker_test YinTrackerTest.cpp)
target_link_libraries(yin_tracker_test PRIVATE wyldtone_yin)
add_test(NAME YinTrackerSynthetic COMMAND yin_tracker_test)

add_executable(bass_synth_control_test BassSynthControlTest.cpp BassSynthControl.cpp)
target_link_libraries(bass_synth_control_test PRIVATE wyldtone_yin)
add_test(NAME BassSynthControlIntegration COMMAND bass_synth_control_test)

add_executable(yin_legato_switch_test LegatoSwitchTest.cpp)
target_link_libraries(yin_legato_switch_test PRIVATE wyldtone_yin)
add_test(NAME YinTrackerLegatoSwitch COMMAND yin_legato_switch_test)
